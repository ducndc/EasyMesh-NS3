\section{Tổng Quan Về Client Steering}
% ============================================================

Trong mạng EasyMesh, các thiết bị khách (STA) có thể kết nối vào bất kỳ điểm truy cập
nào trong tầm phủ sóng, dẫn đến nguy cơ phân bổ tải không đồng đều hoặc STA bám víu
vào BSS có tín hiệu kém. Cơ chế \textbf{Client Steering} giải quyết vấn đề này bằng
cách cho phép Controller hoặc Agent chủ động hướng STA sang BSS tối ưu hơn --- về cả
chất lượng tín hiệu (RCPI), tải kênh (Channel Utilization) và băng tần phù hợp.

Steering được áp dụng trong hai ngữ cảnh:
\begin{itemize}
	\item \textbf{Fronthaul Steering:} Điều phối kết nối của thiết bị người dùng (STA)
	giữa các BSS trong mạng, phục vụ chuyển vùng liền mạch và cân bằng tải.
	\item \textbf{Backhaul Steering:} Điều phối liên kết Backhaul giữa các Agent,
	tối ưu hóa cấu trúc topo mạng lưới Mesh khi điều kiện RF thay đổi.
\end{itemize}

\begin{algorithm}[H]
	
	% ---------- Cài đặt hiển thị ----------
	\DontPrintSemicolon
	\SetAlgoLined
	\LinesNumbered                    % đánh số dòng
	\SetNlSty{textbf}{}{:}            % số dòng đậm, kết thúc bằng dấu hai chấm
	\SetAlgoNlRelativeSize{-1}        % cỡ số dòng nhỏ hơn một chút
	
	% ---------- Tiêu đề & I/O ----------
	\caption{Thuật toán chuyển vùng trong mạng EasyMesh}
	\label{alg:easymesh-roaming}
	
	\KwIn{%
		Thông tin STA: kênh hoạt động, RSSI uplink ($\mathrm{RSSI_{ul}}$),
		khả năng hỗ trợ chuẩn 802.11k/v%
	}
	\KwOut{%
		Quyết định chuyển vùng (gửi BTM Request) hoặc kết thúc tiến trình%
	}
	% ========== Bước 1: Thu thập thông tin ==========
	Cập nhật kênh hoạt động, $\mathrm{RSSI_{ul}}$ và trạng thái của STA\;

	% ========== Kiểm tra điều kiện cần ==========
	\eIf{Số lượng AP $> 1$}{
		
		\eIf{$\mathrm{RSSI_{ul}} <$ Ngưỡng chuyển vùng $(\delta)$}{
			% ========== Bước 2: Thu thập thông tin BSS lân cận ==========
			\eIf{STA hỗ trợ 802.11k}{
				Gửi 11k Beacon Request tới STA\;
				Đợi 11k Beacon Report từ STA\;
				\eIf{Nhận được 11k Beacon Report hợp lệ}{
				}{ % 11k thất bại
					Thực hiện Air Monitor để quét BSS lân cận\;
				}
			}{ % STA không hỗ trợ 802.11k
				Thực hiện Air Monitor để quét BSS lân cận\;
			}
			% ========== Bước 3: Chọn BSS mục tiêu ==========
			Tìm kiếm và chọn BSS mới phù hợp nhất\;
			\eIf{Tồn tại BSS mới phù hợp}{
				% ========== Bước 4: Gửi BTM Request và xử lý phản hồi ==========
				$retry \leftarrow 0$\;
				$\mathit{success} \leftarrow \texttt{false}$\;
				
				\While{$\neg\,\mathit{success}$}{
					Tạo và gửi BTM Request tới STA\;
					Đợi BTM Response từ STA\;
					
					\eIf{$\mathit{Status\,Code} = 0$}{
						$\mathit{success} \leftarrow \texttt{true}$\;
						\KwRet{Chuyển vùng thành công\tcp*[f]{STA chấp nhận BTM}}
					}{ % STA từ chối hoặc không phản hồi
						$retry \leftarrow retry + 1$\;
						\eIf{$retry \leq 2$}{
							Đợi $3$ giây trước khi thử lại\;
						}{ % Đã thử lại tối đa 2 lần
							Cấm STA thực hiện chuyển vùng trong thời gian $T_{\mathrm{ban}}$\;
							\KwRet{Chuyển vùng thất bại\tcp*[f]{Vượt quá số lần thử lại}}
						}
					}
				}
				
			}{ % Không tìm thấy BSS phù hợp
				\KwRet{Kết thúc\tcp*[f]{Không tìm thấy BSS phù hợp}}
			}
			
		}{ % RSSI >= ngưỡng
			\KwRet{Kết thúc\tcp*[f]{$\mathrm{RSSI_{ul}} \geq \delta$, không cần chuyển vùng}}
		}
		
	}{ % Số AP <= 1
		\KwRet{Kết thúc\tcp*[f]{Không đủ AP để thực hiện chuyển vùng}}
	}
	
\end{algorithm}

\subsection{Steering Mandate (Lệnh Bắt Buộc)}
% ------------------------------------------------------------

Controller ra lệnh trực tiếp cho Agent phải di chuyển một STA cụ thể sang BSS mục
tiêu được chỉ định. Agent \textbf{bắt buộc} phải thực hiện trong khoảng thời gian cho
phép (Steering Window) và báo cáo kết quả qua bản tin \texttt{Client Steering BTM
	Report}. Nếu STA từ chối BTM Request, Agent có thể dùng biện pháp mạnh hơn là
Disassociation để buộc STA tìm BSS mới.

Steering Mandate thường được dùng khi:
\begin{itemize}
	\item STA bám víu BSS quá tải hoặc có tín hiệu kém và không tự chuyển vùng.
	\item Controller cần tái cân bằng tải khẩn cấp giữa các BSS.
	\item Cần di chuyển STA sang băng tần cao hơn (Band Steering từ 2.4 GHz lên 5/6 GHz).
\end{itemize}


% ------------------------------------------------------------
\subsection{Steering Opportunity (Cơ Hội Điều Phối)}
% ------------------------------------------------------------

Controller cấp cho Agent một \textbf{khoảng thời gian hiệu lực} (Steering Window) và
danh sách STA cần xem xét điều phối, nhưng \textit{không chỉ định} BSS mục tiêu cụ
thể. Agent được toàn quyền đánh giá điều kiện RF thực tế và tự quyết định có nên thực
hiện Steering hay không, cho ai và sang BSS nào.

Phương thức này phù hợp khi Agent có thông tin RF cục bộ tốt hơn Controller (ví dụ:
đo lường kênh theo thời gian thực) và khi mạng cần phản ứng nhanh mà không chờ
vòng lặp điều phối từ Controller.


% ------------------------------------------------------------
\subsection{Agent-initiated Steering (Tác Nhân Chủ Động)}
% ------------------------------------------------------------

Agent \textbf{không cần chờ lệnh} từ Controller để thực hiện Steering. Dựa trên
\textit{Steering Policy TLV} mà Controller đã phân phối trước đó, Agent liên tục
giám sát chất lượng tín hiệu của các STA đang kết nối. Khi RCPI của một STA giảm
xuống dưới ngưỡng cho phép, Agent tự khởi tạo quy trình BTM hoặc Disassociation.

Cơ chế phân quyền này mang lại hai lợi ích quan trọng:
\begin{itemize}
	\item \textbf{Giảm độ trễ phản ứng:} Agent hành động ngay khi phát hiện tín hiệu
	kém, không cần chờ chu kỳ báo cáo lên Controller và nhận lệnh điều phối.
	\item \textbf{Giảm tải Controller:} Controller không cần theo dõi RCPI từng STA
	trên toàn mạng mà chỉ thiết lập chính sách một lần qua Steering Policy TLV.
\end{itemize}

\textbf{Giải thích luồng bản tin:}
\begin{enumerate}
	\item Controller gửi \textit{Client Steering Request} chế độ Mandate, chỉ định
	rõ STA cần điều phối, BSSID mục tiêu và Steering Window (thời gian tối đa
	để hoàn tất chuyển vùng).
	\item Agent gửi khung \textit{BTM Request} (802.11v) đến STA kèm Neighbor Report
	liệt kê BSS ứng viên theo thứ tự ưu tiên.
	\item STA phản hồi \textit{BTM Response} với Status Code: \texttt{0 = Accept}
	(sẽ chuyển vùng) hoặc \texttt{1 = Reject} (từ chối, có thể kèm BSSID
	thay thế mà STA đề xuất).
	\item[\textbf{4a.}] Nếu STA chấp nhận, STA tự ngắt kết nối và thực hiện
	(Re)Association với BSS mục tiêu.
	\item[\textbf{4b*.}] Nếu STA từ chối và chế độ là Mandate, Agent gửi khung
	\textit{Deauthentication} để buộc STA tìm BSS mới.
	\item Agent gửi \textit{Client Steering BTM Report} về Controller mang Status Code
	từ phản hồi STA --- Controller dùng thông tin này để đánh giá hiệu quả
	chiến lược và điều chỉnh chính sách nếu cần~[17.2.30].
\end{enumerate}

\section{Quản Lý Chính Sách Điều Hướng (Steering Policy)}
% ============================================================

Controller đồng bộ toàn bộ cấu hình điều hướng tới các Agent thông qua bản tin
\textbf{Multi-AP Policy Config Request} mang \textit{Steering Policy TLV}~[17.2.11].
Đây là công cụ chính để Controller thiết lập ``luật vận hành'' cho các Agent tự ra
quyết định Steering mà không cần hỏi lại Controller mỗi lần.

Ba tham số cốt lõi trong Steering Policy:
\begin{itemize}
	\item \textbf{RCPI Steering Threshold} (\textit{Ngưỡng tín hiệu}): Giá trị RCPI
	tối thiểu (dBm) mà dưới ngưỡng này Agent phải xem xét chuyển vùng STA sang
	BSS có tín hiệu tốt hơn. Ngưỡng này được đặt riêng cho từng băng tần để phù
	hợp với đặc tính lan truyền sóng khác nhau.
	\item \textbf{Channel Utilization Threshold} (\textit{Ngưỡng tải kênh}): Mức độ
	bận của kênh (tính theo phần trăm thời gian kênh bị chiếm dụng) mà khi vượt
	quá, Agent kích hoạt Band Steering --- hướng STA sang băng tần ít tắc nghẽn
	hơn (ví dụ: từ 2.4\,GHz sang 5\,GHz hoặc 6\,GHz).
	\item \textbf{Steering Disallowed List} (\textit{Danh sách ngoại lệ}): Danh sách
	các STA \textit{không được} tự động điều phối bởi Agent --- chỉ được Steering
	khi có lệnh Mandate trực tiếp từ Controller. Thường áp dụng cho các thiết bị
	nhạy cảm với gián đoạn kết nối như camera IP, thiết bị IoT công nghiệp hoặc
	thiết bị y tế.
\end{itemize}

\textbf{Giải thích luồng bản tin:}
\begin{enumerate}
	\item Controller gửi \textit{Backhaul Steering Request} đến Agent cần di chuyển,
	chỉ định BSSID của Parent mới, kênh mục tiêu và RUID của radio bSTA cần
	thực hiện chuyển vùng.
	\item Agent ngắt liên kết Backhaul hiện tại với Parent cũ.
	\item bSTA quét, tìm và thực hiện (Re)Association với BSSID của Parent Agent mới.
	\item Agent gửi \textit{Backhaul Steering Response} về Controller xác nhận kết quả:
	thành công kèm BSSID thực tế đang kết nối, hoặc thất bại kèm lý do (không
	tìm thấy BSSID mục tiêu, xác thực thất bại).
	\item Agent gửi \textit{1905 Topology Notification} thông báo thay đổi liên kết
	Backhaul, giúp Controller cập nhật bản đồ topo mạng Mesh.
\end{enumerate}

\section{Các Điểm Kỹ Thuật Then Chốt}
% ============================================================

\begin{itemize}
	\item \textbf{Steering Policy TLV~[17.2.11]:} Là công cụ chính để Controller
	thiết lập ``luật vận hành'' cho các Agent tự ra quyết định Steering. Bằng
	cách định nghĩa ngưỡng RCPI, ngưỡng tải kênh và danh sách ngoại lệ một lần
	duy nhất, Controller ủy quyền cho Agent hành động tự chủ mà không cần can
	thiệp liên tục --- giúp giảm đáng kể lưu lượng báo hiệu trên mạng 1905.1.
	
	\item \textbf{BTM Status Code~[17.2.30]:} Sau mỗi lần Steering, Agent báo cáo lại
	mã trạng thái từ phản hồi BTM của STA: \texttt{0 = Accept} (STA đồng ý
	chuyển vùng), \texttt{1 = Reject} (STA từ chối, có thể kèm BSSID thay thế
	mà STA đề xuất), hoặc các mã lỗi khác. Controller dùng lịch sử Status Code
	này để đánh giá tỉ lệ thành công của từng chiến lược Steering và tự động
	điều chỉnh ngưỡng RCPI hoặc danh sách BSS ứng viên cho phù hợp.
	
	\item \textbf{Client Association Control cho Legacy STA~[11.6]:} Đối với các STA
	đời cũ không hỗ trợ 802.11v BTM, hệ thống sử dụng cơ chế gián tiếp: (1)~gửi
	lệnh Block để ngăn STA kết nối lại BSS hiện tại, (2)~ngắt kết nối cưỡng bức
	bằng Disassociation, (3)~buộc STA quét và tìm BSS mới, (4)~gửi lệnh Unblock
	sau khi STA đã chuyển thành công. Tuy gây gián đoạn kết nối ngắn nhưng đây
	là biện pháp duy nhất có hiệu quả với thiết bị Legacy không hỗ trợ chuẩn
	chuyển vùng chủ động.
\end{itemize}
